    Test runs over time

    Pass Target fail Other (crash, timeout)
    Query
    SELECT time_bucket('10 minutes', time) AS bucket, outcome, count(*)
FROM runs
WHERE time > now() - interval '24 hours' AND NOT cached
GROUP BY bucket, outcome
ORDER BY bucket;

    Runs by stage

    runs · avg time
    Query
    SELECT stage, sum(test_runs) AS runs,
       sum(duration_ms) / 1000.0 / nullif(sum(test_runs), 0) AS avg_s
FROM runs_by_stage            -- continuous aggregate
WHERE bucket > now() - interval '24 hours'
GROUP BY stage
ORDER BY runs DESC;

    Pruning vs blind bisect

    with pruning blind
    Query
    SELECT a.analysis_id,
       count(*) FILTER (WHERE r.stage IN ('prune','bisect')
                          AND NOT r.cached)  AS pruned_runs,
       max(a.blind_runs)                     AS blind_runs
FROM analyses a JOIN runs r USING (analysis_id)
WHERE r.time > now() - interval '24 hours'
GROUP BY a.analysis_id;
